// --- レベルアップ版の影生成・更新ロジックの概念 --- function updateShadowGeometry(wallDataList, sunDir, sunAlt, shadowMesh, geometry) { if (sunAlt <= 0 || wallDataList.length === 0) { shadowMesh.visible = false; return; } shadowMesh.visible = true; // 太陽ベクトルから影の伸びる方向を計算 const lengthXZ = Math.sqrt(sunDir.x * sunDir.x + sunDir.z * sunDir.z); const dirX = lengthXZ > 0 ? -sunDir.x / lengthXZ : 0; const dirZ = lengthXZ > 0 ? -sunDir.z / lengthXZ : 0; const shadowLenFactor = 1.0 / Math.tan(sunAlt); let positions = []; wallDataList.forEach(wall => { // 壁の底面4隅の座標を基準に、上部頂点から地面(Y=0.01)への投影交点を正確に計算 const h = wall.height; const w = wall.width / 2; const d = wall.depth / 2; const rotY = wall.rotation || 0; // 向きを考慮 // 簡易的な底面からトップへの投影頂点計算 // 壁の底面手前・奥の4点から、太陽方向へオフセットした地面上の4点を算出 const baseOffsetX = dirX * (h * shadowLenFactor); const baseOffsetZ = dirZ * (h * shadowLenFactor); const p1 = [wall.x - w, 0.02, wall.z - d]; const p2 = [wall.x + w, 0.02, wall.z - d]; const p3 = [wall.x + w + baseOffsetX, 0.02, wall.z - d + baseOffsetZ]; const p4 = [wall.x - w + baseOffsetX, 0.02, wall.z - d + baseOffsetZ]; // 三角形ポリゴンとしてpositions配列に追加（フェンスや複数壁の統合描画に対応） positions.push(...p1, ...p2, ...p3); positions.push(...p1, ...p3, ...p4); }); // BufferGeometryの頂点バッファを直接一括更新することで、DOM負荷とガベージコレクションをゼロに抑制 geometry.setAttribute('position', new THREE.Float32BufferAttribute(positions, 3)); geometry.attributes.position.needsUpdate = true; }
